\begin{abstract}
This document specifies the geometric theorem formalized by this repository.
It separates the input data and newly defined predicates from the theorems
which construct their witnesses. The accompanying proof blueprint gives the
mathematical argument. The scope is a smooth real Liouville structure with
complete flow and an explicit full dynamical skeleton.
\end{abstract}

\section{Input data and the underlying space}
Let $d\geq0$ and let $B=(B_{ij})\in M_d(\Z)$ be skew-symmetric with
$\det B=1$. Choose a set $M\subseteq\{1,\ldots,d\}$ of mutable indices;
write $F$ for its complement. The directed graph on $M$, with an arrow
$i\to j$ when $B_{ij}>0$, is required to be acyclic. There is no
acyclicity assumption on the full graph. The determinant condition concerns
the entire matrix, including frozen indices.

For $a\in\C^d$ put
\[
 P_i(a)=\prod_{j=1}^d a_j^{[B_{ij}]_+},\qquad
 Q_i(a)=\prod_{j=1}^d a_j^{[-B_{ij}]_+},\qquad [n]_+=\max(n,0).
\]
Define the actual space of complex exchange points
\[
 X_B=\left\{(a,b)\in\C^d\times\C^M:\
 a_ib_i=P_i(a)+Q_i(a)\ (i\in M),\quad a_f\ne0\ (f\in F)\right\}.
\]
It has the subspace topology from the indicated vector space. The frozen
coordinates are variable points of $\C^*$, not specialized coefficients.
The point $a_j=1$, $b_i=2$ shows that this space is nonempty.
Its initial torus is $T=\{a_1\cdots a_d\ne0\}\subseteq X_B$.
\leanref{ClusterSkeleton/Seed.lean}{Seed, Seed.Unimodular, Seed.Point}

\section{Two independent subsets}
\begin{definition}[Coordinate cap]
The coordinate cap is the subset
\[
 C_B=\left\{(a,b)\in X_B:\
 |a_f|=1\ (f\in F),\quad
 |a_i|\leq1,\ (1-|a_i|^2)b_i=0\ (i\in M)\right\}.
\]
\end{definition}
In particular, $|a_i|<1$ forces $b_i=0$. At $|a_i|=1$ the product
condition imposes no further restriction on $b_i$; the exchange equation
still applies. This definition contains no vector field or flow.
Compactness of $C_B$ is a theorem, not a field supplied with the definition.
\leanref{ClusterSkeleton/CapSet.lean}{coordinateCap, isCompact_coordinateCap}

\begin{definition}[Reference strata]
Call $I\subseteq M$ \emph{independent} if $B_{ij}=0$ for all $i,j\in I$, and
for $I\subseteq\{1,\ldots,d\}$ put
\[
 \Lambda_I=\left\{(a,b)\in X_B:\
 b_i=0,\ |a_i|\leq1\ (i\in M\cap I),\quad |a_j|=1\ (j\notin I)\right\}.
\]
\end{definition}
Each $\Lambda_I$ is closed, and compact when $I\subseteq M$. For independent
$I$ this is the reference's stratum. The cap is their union,
\[
 C_B=\bigcup_{I\subseteq M\text{ independent}}\Lambda_I,
\]
because a cap point lies in the stratum indexed by its own set of strict disk
directions $\{i\in M:|a_i|<1\}$, which is independent, while every
$\Lambda_I$ with $I\subseteq M$ lies in $C_B$. Independence is needed for
neither inclusion; it only indexes the union by the reference's family.
\leanref{ClusterSkeleton/CapStrata.lean}{lambdaStratum, lambdaStratum_subset_coordinateCap, coordinateCap_subset_lambdaStratum_strictCapIndices, coordinateCap_eq_iUnion_lambdaStratum}

\begin{definition}[Dynamical skeleton convention]
For a continuous real flow $\Phi$ on $X_B$, let
\[
 \Skel(\Phi)=\left\{x\in X_B:
 \overline{\{\Phi_t(x):t\geq0\}}^{\,X_B}\ \text{is compact}\right\}.
\]
\end{definition}
The closure is taken in the actual space $X_B$. This is the
\emph{relatively compact forward orbit} convention. It does not assert that
points are fixed, and it does not use $C_B$ in its definition.
\leanref{ClusterSkeleton/TrappedSet.lean}{trappedSet}

\section{The geometric theorem}
\begin{theorem}[Formalized target]\label{thm:target}
For every input $(d,B,M)$ above, there exist a smooth real manifold
structure on the fixed topological space $X_B$ of real dimension $2d$,
a smooth real one-form $\lambda$, a smooth vector field $Z$, and a jointly
smooth flow $\Phi:\R\times X_B\to X_B$, with the following properties.
\begin{enumerate}
\item The actual inclusion $X_B\hookrightarrow\C^d\times\C^M$ is a smooth
immersion, and $T$ is dense in $X_B$.
\item $\omega=\dd\lambda$ is nondegenerate, and on $T$ it satisfies
\[
 \omega=\im\!\left(\frac12\sum_{i,j=1}^d
 B_{ij}\,\dd\log a_i\wedge\dd\log a_j\right).
\]
The differential $\dd\log a_i$ means $\dd a_i/a_i$; no global logarithm
is chosen. Since $\omega=\dd\lambda$, the form is closed.
\item $\iota_Z\omega=\lambda$. For every $x\in X_B$ and $t\in\R$,
\[
 \Phi_0(x)=x,\qquad \Phi_{t+s}(x)=\Phi_t(\Phi_s(x)),\qquad
 \frac{\dd}{\dd t}\Phi_t(x)=Z(\Phi_t(x)).
\]
Thus this same vector field is complete in both time directions.
\item For this same flow, there is literal equality of subsets of $X_B$:
\[
 \boxed{\Skel(\Phi)=C_B
 =\bigcup_{I\subseteq M\text{ independent}}\Lambda_I.}
\]
\end{enumerate}
\end{theorem}

The second equality is the reference's stratified form of the identity. It is
a restatement of the same proved equality through the cap description above,
not a further claim.
\leanref{ClusterSkeleton/FullSkeleton.lean}{trappedSet_eq_coordinateCap, trappedSet_eq_iUnion_lambdaStratum}

The quantifier order is
\[
 \forall(d,B,M)\text{ as above},\quad\exists(\text{smooth structure},
 \lambda,Z,\Phi)\quad\text{with all four properties}.
\]
Neither a flow, completeness, compactness of the cap, nor either skeleton
inclusion is an additional hypothesis. The construction uses positive cutoff
parameters $\varepsilon,\delta$ and $\kappa\geq2\varepsilon+1$.
The final existence theorem chooses $\varepsilon=\delta=1$, $\kappa=3$;
these are auxiliary parameters, not values of frozen coordinates.

\section{Scope and verification}
This is the Liouville/skeleton part of the original acyclic-cluster
geometric problem. The statement does not include a holomorphic extension
theorem for the complex logarithmic two-form, finite type, a product
description of individual cap pieces, generalized cocores, or any Fukaya,
microlocal, or homological mirror symmetry equivalence.

The repository's \lean{verification/README.md} distinguishes historical
Horizon source-bound acceptance from the publication checkout's reproducible
checks. The Lean kernel checks the supplied proofs; a separate statement
review checks that the definitions and quantifiers express
Theorem~\ref{thm:target}. The accepted foundation allowlist is
\lean{propext}, \lean{Classical.choice}, and \lean{Quot.sound}; these are
standard logical foundations, not additional geometric assumptions.


\section{Updated reference and continuation scope}
The accompanying proof blueprint now follows the supplied appendix
``Skeleton in the acyclic case''. It uses column exchange exponents and
an integral unimodular completion $E$ with a positive rational diagonal
$\widetilde D$ such that $E\widetilde D$ is skew-symmetric.
The existing formalized theorem above is the specialization
$\widetilde D=I$. Row and column exchange conventions interchange the
two monomials in this specialization; their sum is unchanged.
The continuation uses the existing code to check the supplied argument in
the requested skew-symmetric case. The weighted notation in the reference
does not certify a general skew-symmetrizable Lean theorem.
Unimodularity and mutable acyclicity remain hypotheses.
